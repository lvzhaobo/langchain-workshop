构建图的代码（demo.py 第143-170行）

def build_demo_graph(store, checkpointer):
    builder = StateGraph(MemoryAgentState)

    # 添加 6 个节点
    builder.add_node("load_memory", load_memory_node)
    builder.add_node("agent", demo_agent_node)
    builder.add_node("detect_changes", detect_profile_changes_node)
    builder.add_node("save_memory", save_to_memory_node)
    builder.add_node("update_profile", update_profile_node)
    builder.add_node("consolidate", consolidate_memory_node)

    # 连接边：定义流转顺序
    builder.set_entry_point("load_memory")
    builder.add_edge("load_memory", "agent")
    builder.add_edge("agent", "detect_changes")

    # 条件边：有画像变更 → update_profile，否则 → save_memory
    builder.add_conditional_edges(
        "detect_changes",
        lambda s: "update_profile"
            if s.get("profile_delta")
            else "save_memory",
        {"update_profile": "update_profile",
         "save_memory": "save_memory"})

    builder.add_edge("update_profile", "save_memory")
    builder.add_edge("save_memory", "consolidate")
    builder.add_edge("consolidate", END)

    return builder.compile(
        checkpointer=checkpointer,
        store=store)

4 如何调用：增/查/改记忆的真实代码

4.1 初始化记忆系统

# demo.py 第187-192行
from langgraph.store.memory import InMemoryStore
from langgraph.checkpoint.memory import MemorySaver
from memory_store import MemoryStoreManager

store = InMemoryStore()              # 长期记忆后端
checkpointer = MemorySaver()         # 短期记忆持久化
memory = MemoryStoreManager(store, "user_001")

生产环境将 InMemoryStore 替换为 AsyncPostgresStore，图代码无需改动。

4.2 写入一条长期记忆

# memory_store.py 第48行
# 写入一条记忆，自动去重（Jaccard ≥ 0.7）
await memory.remember(
    content="用户是后端工程师，主要使用 Python 和 TypeScript",
    category="knowledge",   # preference/knowledge/decision/pattern
    score=8.0,              # 重要性评分 1-10
    confidence=0.9,          # 置信度 0.0-1.0
)
# → Store.put(("memories", "user_001"), fact_id, {...})

4.3 语义搜索记忆

# memory_store.py 第103行
# 语义搜索，后处理：时间衰减 + 回忆加权
results = await memory.recall(
    query="SQL 注入",
    limit=5,              # 返回前 5 条
    min_score=1.0,         # 最低评分过滤
)
# → Store.search(("memories", "user_001"), query, limit)
# 每条结果: {content, category, score, effective_score, ...}

4.4 更新用户画像（EMA 平滑）

# memory_store.py 第210行
# EMA 平滑：new = old×0.7 + new×0.3
await memory.update_profile(
    trait_updates={
        "security_sensitivity": "high",
        "preferred_language": "TypeScript",
    },
    confidence=0.7,
)
# → Store.put(("profiles", "user_001"), "profile", {...})
💡 记忆的生命周期是怎么触发的？ 不是手动调用！在图的执行流程中，save_to_memory_node 在每次 Agent 响应后自动执行，提取事实并写入 Store。你只需要配置好图，记忆自动积累。
5 完整调用流程（来自 demo.py）

5.1 启动 → 对话 → 记忆持久化

# demo.py main() 函数

# Step 0: 预填充已有记忆（模拟历史对话）
memory.remember("用户是后端工程师，主要技术栈 Python 和 TypeScript",
                category="knowledge", score=8.0, confidence=0.9)
memory.update_profile({"security_sensitivity": "high"}, confidence=0.8)

# Step 1: 构建带记忆的 Agent 图
app = build_demo_graph(store, checkpointer)

# Step 2: 第 1 天——用户发起审查
config1 = {"configurable": {"thread_id": "session-day-1", "store": store}}
result1 = await app.ainvoke(
    {"messages": [HumanMessage(content="帮我审查代码，我比较关注SQL注入安全")],
     "session_meta": {"user_id": "user_001", "session_id": "session-day-1"}},
    config1,
)
# ↑ 此行自动执行了 load_memory → agent → detect_changes → update_profile → save_memory → consolidate

# Step 3: 第 2 天——新会话，自动恢复记忆
config2 = {"configurable": {"thread_id": "session-day-2", "store": store}}
result2 = await app.ainvoke(
    {"messages": [HumanMessage(content="又有一批新代码需要审查")],
     "session_meta": {"user_id": "user_001", "session_id": "session-day-2"}},
    config2,
)
# ↑ load_memory 节点自动从 Store 恢复了第1天的画像和偏好！

# Step 4: 搜索验证
results = memory.recall("SQL 注入", limit=3)
profile = memory.get_profile()
print(profile["traits"])  # → {"security_sensitivity": {"value": "high", "confidence": 0.7}, ...}

5.2 关键：新会话如何自动恢复记忆？

load_memory_node 做的事（memory_graph.py:40） 1. 从 Store 获取用户画像
2. 用当前用户消息搜索相关长期记忆
3. 组装为 SystemMessage 注入到 messages
→ Agent 看到的上下文自动包含了历史记忆
为什么新 thread_id 也能恢复？ Store 是跨 thread 的（Checkpointer 是 thread 内），所有会话共享同一个 Store 实例。只要 user_id 相同，就能查到同一份记忆。
6 画像、SOUL.md、Memory 三者的区别
概念WHO（关于谁）WHAT（存什么）变化频率LangGraph 存储位置
SOUL.md Agent 自己 核心哲学、行为准则、语气、身份 几乎不变 Store namespace ("agent", "persona")
IDENTITY.md Agent 自己 表面身份：名称、代词、默认语言 几乎不变 Store namespace ("agent", "identity")
USER.md / 用户画像 用户 角色、技能、偏好、沟通风格、审查优先级 缓慢进化 (EMA) Store namespace ("profiles", user_id)
MEMORY.md / 长期记忆 对话内容 从对话中提取的事实：技术栈、决策、偏好表述 持续增长 Store namespace ("memories", user_id)
情节记忆 会话内容 每日会话摘要、关键决策记录 每天新增 Store namespace ("sessions", user_id, date)
简单记忆法： SOUL.md = Agent 的"人格"（我是谁）
画像 = Agent 对用户的"认知模型"（你是谁）
长期记忆 = Agent 的"经验"（我们之间发生了什么）
三者都存在同一个 Store 实例中，namespace 隔离。
7 完整 API 速查（MemoryStoreManager 类）
方法参数作用对应 Store API
remember() content, category, score, confidence 写入一条记忆（自动去重） store.put(("memories", uid), fact_id, fact)
recall() query, limit, min_score 语义搜索记忆（时间衰减+加权排序） store.search(("memories", uid), query, limit)
forget() fact_id 标记记忆为过时（不物理删除） store.put(... superseded_by="manual_forget")
get_profile() user_id 获取完整用户画像 store.get(("profiles", uid), "profile")
update_profile() trait_updates, confidence EMA 平滑更新画像特征 store.put(("profiles", uid), "profile", {...})
add_profile_fact() fact, confidence 添加画像事实（Jaccard 去重） store.put(("profiles", uid), "profile", {...})
save_session_summary() summary, session_id 保存一日会话摘要（情节记忆） store.put(("sessions", uid, date), session_id, {...})
set_agent_memory() key, value 设置 Agent 自身记忆（类似 SOUL.md） store.put(("agent", key), "current", value)
get_agent_memory() key 获取 Agent 自身记忆 store.get(("agent", key), "current")
8 从演示到生产（代码无需改动）
阶段Store 后端Checkpointer切换方式
1. 演示/测试 InMemoryStore() MemorySaver() 当前 demo.py
2. 本地持久化 SqliteStore SqliteSaver 换两行 import
3. 生产部署 AsyncPostgresStore AsyncPostgresSaver 换两行 import + 连接字符串
关键设计原则： 图的构建代码（build_demo_graph）完全不感知底层存储。换 InMemoryStore → AsyncPostgresStore 后，所有记忆自动持久化到 PostgreSQL，图和节点函数零改动。
