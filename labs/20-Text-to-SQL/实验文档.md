# Lab 20 · Text-to-SQL —— 自然语言查询设备数据

> 知识库能搜文档了，但设备的结构化数据（设备表、告警表）还在用固定的 API 查询——“查告警设备有几台”这种灵活问题 Agent 答不了。本实验给 Agent 装上 SQL 工具：自然语言 → SQL → 数据库查询 → 返回结果。Agent 从此能"说出数据"。

---

## 🎯 实验目标

- 把 `devices.json` 导入 SQLite 内存数据库，建两张表（devices + alarms）
- 给 Agent 新增 `query_devices_sql` 工具——输入自然语言，输出查询结果
- Agent 自动生成 SQL、执行、解释结果
- 新增后端接口 `/sql/query` 和前端按钮「📊 智能问数」

## 📦 标准路径

### 第一步：把设备数据导入 SQLite

```python
# text_to_sql.py
import json
import sqlite3
from pathlib import Path


def init_db():
    """把 devices.json 导入 SQLite 内存数据库"""
    conn = sqlite3.connect(":memory:")
    conn.row_factory = sqlite3.Row  # 让查询结果可以用列名访问

    # 建表
    conn.execute("""
        CREATE TABLE devices (
            device_id TEXT PRIMARY KEY,
            model TEXT,
            vendor TEXT,
            site TEXT,
            type TEXT,
            status TEXT,
            ip TEXT,
            online_date TEXT
        )
    """)
    conn.execute("""
        CREATE TABLE alarms (
            alarm_id TEXT PRIMARY KEY,
            device_id TEXT,
            type TEXT,
            severity TEXT,
            time TEXT,
            description TEXT,
            FOREIGN KEY (device_id) REFERENCES devices(device_id)
        )
    """)

    # 导入数据
    with open("data/devices.json", encoding="utf-8") as f:
        devices = json.load(f)

    for d in devices:
        conn.execute(
            """INSERT INTO devices
               (device_id, model, vendor, site, type, status, ip, online_date)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?)""",
            (d["device_id"], d["model"], d["vendor"], d["site"],
             d["type"], d["status"], d["ip"], d["online_date"])
        )
        for a in d.get("alarms", []):
            conn.execute(
                """INSERT OR IGNORE INTO alarms
                   (alarm_id, device_id, type, severity, time, description)
                   VALUES (?, ?, ?, ?, ?, ?)""",
                (a["alarm_id"], d["device_id"], a["type"],
                 a["severity"], a["time"], a["description"])
            )

    conn.commit()
    return conn
```

### 第二步：实现 NL → SQL 工具

```python
from langchain_openai import ChatOpenAI
from langchain.agents import tool


@tool
def query_devices_sql(question: str) -> str:
    """用自然语言查询设备数据库。当用户问统计类问题（"有几台""哪个厂商最多"）
    或复杂筛选（"P0 告警的华为设备"）时使用此工具。
    
    输入要查询的问题（自然语言），工具自动翻译为 SQL 并执行。
    """
    # 获取表结构供 LLM 参考
    schema = """
    表 devices: device_id(TEXT), model(TEXT), vendor(TEXT), site(TEXT),
               type(TEXT), status(TEXT), ip(TEXT), online_date(TEXT)
    表 alarms: alarm_id(TEXT), device_id(TEXT), type(TEXT),
              severity(TEXT), time(TEXT), description(TEXT)
    
    status 可选值: running, alarm, maintenance
    severity 可选值: P0, P1, P2, P3
    vendor 可选值: 华为, 中兴, 爱立信, 诺基亚
    """

    # 让 LLM 把自然语言翻译为 SQL
    llm = ChatOpenAI(
        model="deepseek-chat", temperature=0,
        api_key="你的 API Key", base_url="https://api.deepseek.com/v1",
    )

    sql_prompt = f"""你是一个 SQL 专家。根据表结构和用户问题，生成 SQLite 的 SELECT 语句。
只输出 SQL，不要解释。

表结构：
{schema}

用户问题：{question}

SQL："""
    
    sql_response = llm.invoke(sql_prompt)
    sql = sql_response.content.strip()

    # 清理 SQL（去掉可能的 markdown 格式）
    if "```sql" in sql:
        sql = sql.split("```sql")[1].split("```")[0]
    elif "```" in sql:
        sql = sql.split("```")[1].split("```")[0]

    # 安全检查：只允许 SELECT
    if not sql.strip().upper().startswith("SELECT"):
        return f"错误：只允许 SELECT 查询。生成的语句是：{sql}"

    # 执行 SQL
    try:
        cursor = db_conn.execute(sql)
        rows = cursor.fetchall()

        if not rows:
            return f"查询成功，但没有匹配的数据。\nSQL: {sql}"

        # 格式化结果
        columns = [desc[0] for desc in cursor.description]
        lines = [f"SQL: {sql}", f"\n结果（{len(rows)} 行）：\n"]

        for row in rows[:20]:  # 最多返回 20 行
            line = " | ".join(f"{col}={row[col]}" for col in columns)
            lines.append(f"  {line}")

        if len(rows) > 20:
            lines.append(f"  ... 还有 {len(rows) - 20} 行未显示")

        return "\n".join(lines)

    except sqlite3.OperationalError as e:
        return f"SQL 执行错误：{str(e)}\n生成的 SQL：{sql}"
```

### 第三步：组装 Agent + 测试

```python
from langchain.agents import create_react_agent, AgentExecutor
from langchain.prompts import ChatPromptTemplate

# 初始化数据库连接
db_conn = init_db()
print(f"导入 {db_conn.execute('SELECT COUNT(*) FROM devices').fetchone()[0]} 台设备")
print(f"导入 {db_conn.execute('SELECT COUNT(*) FROM alarms').fetchone()[0]} 条告警")

# 创建 Agent
tools = [query_devices_sql]

prompt = ChatPromptTemplate.from_messages([
    ("system", """你是 5G 基站数据分析助手。你可以使用 query_devices_sql 工具查询设备数据库。

查询技巧：
- 统计类问题（"有几台""哪个最多"）→ 用 query_devices_sql
- 复杂筛选（"P0 告警的华为设备"）→ 用 query_devices_sql
- 设备详情查询 → 用 query_device（另一个工具）

可用工具：{tools}
工具名称：{tool_names}

格式：Question → Thought → Action → Action Input → Observation → Final Answer"""),
    ("human", "{input}"),
    ("placeholder", "{agent_scratchpad}"),
])

llm = ChatOpenAI(model="deepseek-chat", temperature=0, ...)
agent = create_react_agent(llm, tools, prompt)
executor = AgentExecutor(
    agent=agent, tools=tools, verbose=True,
    max_iterations=8, handle_parsing_errors=True,
)

# ── 测试 ──
test_queries = [
    "告警设备有多少台？按厂商分组统计",
    "P0 级别告警有几条？分别是什么类型？",
    "华为设备中状态为 running 的有多少？",
    "哪个站点的告警设备最多？列出站点名和数量",
    "爱立信设备中，告警最多的型号是什么？",

for query in test_queries:
    print(f"\n{'='*60}")

### 第四步：接入平台

在 `server.py` 中新增 `/sql/query` 接口：

```python
# server.py 新增
import sqlite3
from text_to_sql import init_db, make_sql_agent

sql_agent = None  # 延迟初始化


def get_sql_agent():
    global sql_agent
    if sql_agent is None:
        sql_agent = make_sql_agent()
    return sql_agent


@app.route("/sql/query", methods=["POST"])
def sql_query():
    """智能问数：自然语言 → SQL → 结果"""
    data = request.get_json()
    question = data.get("question", "").strip()
    if not question:
        return jsonify({"reply": "请输入查询问题"}), 400

    try:
        agent = get_sql_agent()
        result = agent.invoke({"input": question})
        return jsonify({"reply": result["output"]})
    except Exception as e:
        return jsonify({"reply": f"查询失败：{str(e)}"})
```

## 🏗️ 进阶挑战

### 1. 防止 SQL 注入 / 危险操作

当前只允许 SELECT。加强：禁止 DROP/DELETE/UPDATE/INSERT/ALTER、禁止 `--` 注释、限制返回行数上限（100 行）。如果检测到危险 SQL，返回明确的拒绝信息。

### 2. 结果可视化

把 SQL 查询结果渲染为前端图表——柱状图展示"各厂商告警设备数"、饼图展示"告警级别分布"。Chart.js 或 ECharts 均可。

### 3. 复杂查询支持

支持多表 JOIN、子查询、聚合函数。如："列出 P0 告警的设备及其站点的告警总数，按告警数降序"——这种查询需要 `JOIN + GROUP BY + ORDER BY + 子查询`。

## ⚠️ 常见问题

| 问题 | 原因 | 解决方法 |
|------|------|---------|
| SQL 语法错误 | LLM 生成了 SQLite 不支持的语法 | 在 schema 中强调"这是 SQLite"，避免 MySQL/PG 特有函数 |
| 查询结果为空 | LLM 对数据值猜测错误 | 在 schema 中提供实际数据分布（vendor 有哪些值等） |
| Agent 没有用 SQL 工具 | Prompt 没引导 | 在 system prompt 中明确"统计/分组/筛选类问题用 SQL 工具" |
| 中文列名或值匹配失败 | SQLite 的字符编码 | 确认所有数据都是 UTF-8 |
| Agent 循环调 SQL | 第一次结果不满意，反复重试 | 设置 `max_iterations=8` 限制 |
